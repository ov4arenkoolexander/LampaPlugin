(function () {
    'use strict';

    console.log('My Lampa Plugin loaded');

    if (window.Lampa) {
        Lampa.Noty.show('Привіт! Плагін працює.');
    }
})();
(function () {
    'use strict';

    function addSource() {

        Lampa.Component.add('my_source', {
            name: 'Моє джерело',

            start: function () {
                Lampa.Noty.show('Джерело запущено');

                let films = [
                    {
                        title: 'Мій перший фільм',
                        year: '2026'
                    },
                    {
                        title: 'Другий фільм',
                        year: '2025'
                    }
                ];

                console.log(films);
            }
        });

    }

    if (window.Lampa) {
        addSource();
        Lampa.Noty.show('Джерело додано!');
    }

})();
